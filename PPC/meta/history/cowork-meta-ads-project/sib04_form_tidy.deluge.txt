// SIB-04 Form Tidy - 10 Oct 2026. Workflow "SIB-04 Form Tidy" (Leads, Create, Meta Lead Form is SIB-04 ACR Form v1).
// 1. Description: LeadChain joins the two form answers with no separator; insert ", ".
// 2. Product Name: "2 Pull Product Info" rebuilds Product Cat from Product Name and blanks it when Product Name is empty.
//    LLP -> LLP Annual Return (ACR); anything else -> ACR - Company Annual Return. Only if blank.
// No workflow re-trigger. Form Q1 options: Private Limited / One Person Company (OPC) / LLP.
// v2 10 Oct 2026 ~11:10: Product Cat set ONLY IF BLANK, so the scheduled re-run (N min after create) never overwrites an RM's category.
rec = zoho.crm.getRecordById("Leads",leadId.toLong());
desc = ifnull(rec.get("Description"),"").trim();
upd = Map();
entity = "";
entities = {"One Person Company (OPC)","Private Limited","LLP"};
for each  e in entities
{
	if(entity == "" && desc.startsWith(e))
	{
		entity = e;
	}
}
if(entity != "")
{
	rest = desc.subString(entity.length()).trim();
	if(rest.startsWith(","))
	{
		rest = rest.subString(1).trim();
	}
	if(rest != "")
	{
		upd.put("Description",entity + ", " + rest);
	}
}
if(rec.get("Product_Name") == null)
{
	if(entity == "LLP")
	{
		upd.put("Product_Name","1123581000080093673");
	}
	else
	{
		upd.put("Product_Name","1123581000000185017");
	}
}
if(rec.get("Product_Category") == null)
{
	upd.put("Product_Category","Annual Filing");
}
if(upd.size() > 0)
{
	opt = Map();
	opt.put("trigger",List());
	r = zoho.crm.updateRecord("Leads",leadId.toLong(),upd,opt);
	info "sib04_form_tidy " + leadId + ": " + r;
}
else
{
	info "sib04_form_tidy " + leadId + ": nothing to change";
}
