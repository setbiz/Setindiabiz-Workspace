// Body (between the braces of the function):

// PPC Form Tidy - one function for every Meta lead form listed in CONFIG. Forms not listed are left alone.
// 1. Description: LeadChain joins the form answers with no separator; insert ", " after the first answer.
// 2. Product Name, only if blank: the product for the first answer, else the form's default product.
// 3. Product Cat and Report Cat, each only if blank: copied from that Products record.
// Updates fire no workflow. Safe to run twice (instant + scheduled).
rec = zoho.crm.getRecordById("Leads",leadId.toLong());
form = ifnull(rec.get("Meta_Lead_Form"),"").trim();
// ===== CONFIG: one block per lead form =====
cfg = Map();
// SIB-04 Annual Filings - form "SIB-04 ACR Form v1" (2635626386872789)
c = Map();
c.put("product","1123581000000185017");
// ACR - Company Annual Return
q1 = {"One Person Company (OPC)","Private Limited","LLP"};
c.put("q1",q1);
byAnswer = Map();
byAnswer.put("LLP","1123581000080093673");
// LLP Annual Return (ACR)
c.put("byAnswer",byAnswer);
cfg.put("SIB-04 ACR Form v1",c);
// SIB-05 India Entry - form "SIB-05 India Entry Form v1" (added 10 Oct 2026)
c = Map();
c.put("product","1123581000000185013");
// FDI - Subsidiary Incorporation (Subsidiary and Not sure yet)
q1 = {"Subsidiary (Indian company)","Branch office","Liaison office","Not sure yet"};
c.put("q1",q1);
byAnswer = Map();
byAnswer.put("Branch office","1123581000057725451");
// Branch Office Registration of Foreign Company
byAnswer.put("Liaison office","1123581000057725457");
// FDI - Liaison Office
c.put("byAnswer",byAnswer);
cfg.put("SIB-05 India Entry Form v1",c);
// ===== end CONFIG =====
if(cfg.containsKey(form) == false)
{
	info "ppc_form_tidy " + leadId + ": form '" + form + "' not in CONFIG, nothing done";
}
else
{
	c = cfg.get(form);
	desc = ifnull(rec.get("Description"),"").trim();
	upd = Map();
	answer = "";
	opts = c.get("q1");
	for each  o in opts
	{
		if(answer == "" && desc.startsWith(o))
		{
			answer = o;
		}
	}
	if(answer != "")
	{
		rest = desc.subString(answer.length()).trim();
		if(rest.startsWith(","))
		{
			rest = rest.subString(1).trim();
		}
		newDesc = answer + ", " + rest;
		if(rest != "" && newDesc != desc)
		{
			upd.put("Description",newDesc);
		}
	}
	pid = "";
	if(rec.get("Product_Name") != null)
	{
		pid = rec.get("Product_Name").get("id");
	}
	else
	{
		pid = c.get("product");
		ba = c.get("byAnswer");
		if(answer != "" && ba.containsKey(answer))
		{
			pid = ba.get(answer);
		}
		upd.put("Product_Name",pid);
	}
	if(rec.get("Product_Category") == null || rec.get("Report_Cat") == null)
	{
		prod = zoho.crm.getRecordById("Products",pid.toLong());
		if(rec.get("Product_Category") == null && prod.get("Product_Category") != null)
		{
			upd.put("Product_Category",prod.get("Product_Category"));
		}
		if(rec.get("Report_Cat") == null && prod.get("Report_Cat") != null)
		{
			upd.put("Report_Cat",prod.get("Report_Cat"));
		}
	}
	if(upd.size() > 0)
	{
		opt = Map();
		opt.put("trigger",List());
		r = zoho.crm.updateRecord("Leads",leadId.toLong(),upd,opt);
		info "ppc_form_tidy " + leadId + " (" + form + "): " + upd.keys() + " -> " + r;
	}
	else
	{
		info "ppc_form_tidy " + leadId + " (" + form + "): nothing to change";
	}
}
