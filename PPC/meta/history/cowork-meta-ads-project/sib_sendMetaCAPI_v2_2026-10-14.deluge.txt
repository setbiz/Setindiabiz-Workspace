void automation.sib_sendMetaCAPI1(String moduleName,String recordId)
{
// v2 14 Oct 2026.
//  1. Paid-Meta sources now include "Advt - WhatsApp" (Leads and Deals).
//  2. Each event is sent once per record. Field Meta_CAPI_Event (Leads and Deals) holds the highest event sent;
//     an event is sent only if it ranks above it: ValidLead 1 < QualifiedLead 2 < Committed 3 < Purchase 4.
//     Before v2 every Stage edit inside the won stages (Won -> W-Login -> W-Finance -> W-SKV ...) sent another Purchase.
//  3. event_id = <recordId>-<eventName>, and custom_data.lead_source.
// Same body in both copies; only the first line differs (Lead copy: sib_sendMetaCAPI, Deal copy: sib_sendMetaCAPI1).
datasetId = "1242417817083521";
apiVersion = "v21.0";
accessToken = zoho.crm.getOrgVariable("meta_capi_access_token");
rec = zoho.crm.getRecordById(moduleName,recordId.toLong());
metaSources = {"Advt - Facebook","Advt - Instagram","Advt - WhatsApp","PPC Meta Insta"};
rank = Map();
rank.put("ValidLead",1);
rank.put("QualifiedLead",2);
rank.put("Committed",3);
rank.put("Purchase",4);
lastEvent = ifnull(rec.get("Meta_CAPI_Event"),"").toString();
eventName = "";
value = 0;
actionSource = "system_generated";
email = "";
phone = "";
src = "";
if(moduleName == "Leads")
{
	src = ifnull(rec.get("Lead_Source"),"").toString();
	if(metaSources.contains(src) == false)
	{
		return;
	}
	// Sent before v2 without a recorded event: unknown which one went, so do not send again.
	if(ifnull(rec.get("Meta_CAPI_Sent"),false) == true && lastEvent == "")
	{
		info "Meta CAPI: lead sent before v2, event unknown - skipped";
		return;
	}
	rmq = ifnull(rec.get("RM_Lead_Quality"),"").toString();
	tlq = ifnull(rec.get("Mkt_Leat_Quality"),"").toString();
	csdq = ifnull(rec.get("CSD_Lead_Quality"),"").toString();
	status = ifnull(rec.get("Lead_Status"),"").toString();
	validSet = {"Valid - RM Working","Valid - NIL RM Action","Valid - Casual RM Action","Valid - Good RM Action"};
	progressSet = {"Negotiation","Proposal Sent","Payment Pending"};
	if(progressSet.contains(status))
	{
		eventName = "QualifiedLead";
	}
	else if(validSet.contains(rmq) || validSet.contains(tlq) || validSet.contains(csdq) || status == "Valid Lead")
	{
		eventName = "ValidLead";
	}
	email = ifnull(rec.get("Email"),"").toString();
	phone = ifnull(rec.get("Phone"),"").toString();
	if(phone == "")
	{
		phone = ifnull(rec.get("Mobile"),"").toString();
	}
}
else if(moduleName == "Deals")
{
	con = rec.get("Contact_Name");
	if(con == null)
	{
		return;
	}
	contact = zoho.crm.getRecordById("Contacts",con.get("id").toLong());
	src = ifnull(contact.get("Lead_Source"),"").toString();
	if(metaSources.contains(src) == false)
	{
		return;
	}
	email = ifnull(contact.get("Email"),"").toString();
	phone = ifnull(contact.get("Phone"),"").toString();
	if(phone == "")
	{
		phone = ifnull(contact.get("Mobile"),"").toString();
	}
	stage = ifnull(rec.get("Stage"),"").toString();
	qualifiedSet = {"Qualified","Follow-Up","Negotiation"};
	committedSet = {"Payment Wait"};
	wonSet = {"Won","W-SKV","W-SPI","W-CSD","W-Finance","W-Login","W-ACR Login"};
	if(wonSet.contains(stage))
	{
		eventName = "Purchase";
		value = ifnull(rec.get("Amount"),0);
	}
	else if(committedSet.contains(stage))
	{
		eventName = "Committed";
	}
	else if(qualifiedSet.contains(stage))
	{
		eventName = "QualifiedLead";
	}
}
if(eventName == "")
{
	return;
}
// Once-per-record guard
if(lastEvent != "" && rank.containKey(lastEvent))
{
	if(rank.get(eventName).toLong() <= rank.get(lastEvent).toLong())
	{
		info "Meta CAPI: " + eventName + " not above last sent " + lastEvent + " - skipped";
		return;
	}
}
userData = Map();
leadId = ifnull(rec.get("leadchain0__Social_Lead_ID"),"").toString();
if(leadId != "")
{
	userData.put("lead_id",leadId);
}
if(email != "")
{
	userData.put("em",zoho.encryption.sha256(email.toLowerCase().trim()));
}
if(phone != "")
{
	phNorm = phone.replaceAll("[^0-9]","");
	if(phNorm.length() == 10)
	{
		phNorm = "91" + phNorm;
	}
	userData.put("ph",zoho.encryption.sha256(phNorm));
}
if(userData.size() == 0)
{
	return;
}
cd = Map();
cd.put("lead_source",src);
if(eventName == "Purchase")
{
	cd.put("value",value);
	cd.put("currency","INR");
}
event = Map();
event.put("event_name",eventName);
event.put("event_time",zoho.currenttime.toString().unixEpoch() / 1000);
event.put("action_source",actionSource);
event.put("event_id",recordId + "-" + eventName);
event.put("user_data",userData);
event.put("custom_data",cd);
dataList = List();
dataList.add(event);
payload = Map();
payload.put("data",dataList);
endpoint = "https://graph.facebook.com/" + apiVersion + "/" + datasetId + "/events?access_token=" + accessToken;
response = invokeurl
[
	url :endpoint
	type :POST
	parameters:payload.toString()
	content-type:"application/json"
];
info "Meta CAPI [" + eventName + "] -> " + response;
received = ifnull(response.get("events_received"),0);
if(received > 0)
{
	upd = Map();
	upd.put("Meta_CAPI_Event",eventName);
	if(moduleName == "Leads")
	{
		upd.put("Meta_CAPI_Sent",true);
	}
	noTrig = Map();
	emptyList = List();
	noTrig.put("trigger",emptyList);
	u = zoho.crm.updateRecord(moduleName,recordId.toLong(),upd,noTrig);
	info "Meta CAPI: marked " + eventName + " -> " + u;
}
}
